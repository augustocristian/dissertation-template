% hyperref's colorlinks (and friends) only take effect if set here, in the
% preamble -- setting them via \hypersetup after \begin{document} is too

% a "Option `colorlinks' has already been used" warning.

  pdfauthor={\@author},
  colorlinks=true,
  linkcolor=uoTeal,
  citecolor=uoTeal,
  urlcolor=uoTeal,
}
\makeatother
